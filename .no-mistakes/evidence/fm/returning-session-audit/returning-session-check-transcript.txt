=== fm-returning-session-check.sh: what a caller sees ===

-- session well below the restart band --
$ fm-returning-session-check.sh <tmp>/warm
verdict=resume band=ok context_tokens=50000 restart_tokens=250000 transcript=<tmp>/ch/.claude/projects/-tmp-tmp-wC56cge2NO-warm/session.jsonl
[exit 0]

-- session past the restart band (must not bare-resume) --
$ fm-returning-session-check.sh <tmp>/hot
verdict=restart-with-carryover band=restart context_tokens=260000 restart_tokens=250000 transcript=<tmp>/ch/.claude/projects/-tmp-tmp-wC56cge2NO-hot/session.jsonl
[exit 0]

-- home that has never run a session: no evidence, still resumable --
$ fm-returning-session-check.sh <tmp>/fresh
verdict=unknown reason=fm-context-usage: no Claude transcript directory at <tmp>/ch/.claude/projects/-tmp-tmp-wC56cge2NO-fresh (non-claude harness, or no session yet)
